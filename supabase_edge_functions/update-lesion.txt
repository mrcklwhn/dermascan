import { createClient } from "npm:@supabase/supabase-js@2.39.0";

// --- Environment Variables ---
const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";

if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY) {
  throw new Error("Missing Supabase configuration.");
}

// --- Supabase Client (Service Role) ---
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
  auth: {
    persistSession: false,
    autoRefreshToken: false,
    detectSessionInUrl: false,
  },
});

// --- CORS Headers ---
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// --- Helper: Response ---
const jsonResponse = (body: boolean, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });

Deno.serve(async (req) => {
  // 1. Handle CORS Preflight
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const authHeader = req.headers.get('Authorization');
    if (!authHeader) {
      return jsonResponse(false, 401);
    }

    // Supabase Client für Auth-Verifizierung erstellen
    const supabaseClient = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    const { data: { user }, error: userError } = await supabaseClient.auth.getUser(
      authHeader.replace("Bearer ", "")
    );

    if (userError || !user) {
      return jsonResponse(false, 401);
    }

    // 2. Parse Request Body
    const { lesionId, locationName, suspicious, wasRemoved, testResult, notes } = await req.json();

    // Strikte Validierung: suspicious und wasRemoved müssen Booleans sein
    if (!lesionId || typeof suspicious !== 'boolean' || typeof wasRemoved !== 'boolean') {
      return jsonResponse(false, 400);
    }

    const doctorId = user.id; // Sender ID

    // 3. Fetch Data & Validation

    // 3.1 Check Doctor and practice assignment
    const { data: doctor, error: doctorError } = await supabase
      .from('doctor')
      .select('practiceid')
      .eq('doctorid', doctorId)
      .maybeSingle();

    if (doctorError || !doctor || !doctor.practiceid) {
      if (doctorError) console.error("DB Fetch Error (Doctor):", doctorError);
      return jsonResponse(false, 403);
    }

    // 3.2 Check Lesion and get associated PatientID
    const { data: lesion, error: lesionError } = await supabase
      .from('lesion')
      .select('patientid')
      .eq('lesionid', lesionId)
      .maybeSingle();

    if (lesionError || !lesion) {
      if (lesionError) console.error("DB Fetch Error (Lesion):", lesionError);
      return jsonResponse(false, 404);
    }

    // 3.3 Check Patient and practice assignment
    const { data: patient, error: patientError } = await supabase
      .from('patient')
      .select('practiceid')
      .eq('patientid', lesion.patientid)
      .maybeSingle();

    if (patientError || !patient) {
      if (patientError) console.error("DB Fetch Error (Patient):", patientError);
      return jsonResponse(false, 404);
    }

    // 3.4 Authorization Check: Same practice?
    if (doctor.practiceid !== patient.practiceid) {
      return jsonResponse(false, 403);
    }

    // 4. Execute Database Update Operation
    const { error: updateError } = await supabase
      .from('lesion')
      .update({
        locationname: locationName,
        suspicious: suspicious,
        wasremoved: wasRemoved,
        testresult: testResult,
        notes: notes
      })
      .eq('lesionid', lesionId);

    if (updateError) {
      console.error("DB Update Error:", updateError);
      return jsonResponse(false, 500);
    }

    // 5. Success
    return jsonResponse(true, 200);

  } catch (e) {
    console.error("Function Error:", e);
    return jsonResponse(false, 500);
  }
});
