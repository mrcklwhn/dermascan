import { createClient } from "npm:@supabase/supabase-js@2.39.0";

// --- Environment Variables ---
const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";

if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY) {
  throw new Error("Missing Supabase configuration.");
}

// --- Supabase Client (Service Role) ---
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
  auth: {
    persistSession: false,
    autoRefreshToken: false,
    detectSessionInUrl: false,
  },
});

// --- CORS Headers ---
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// --- Helper: Response ---
const jsonResponse = (body: any, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });

Deno.serve(async (req) => {
  // 1. Handle CORS Preflight
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const authHeader = req.headers.get('Authorization');
    if (!authHeader) {
      return jsonResponse({ error: "No authorization header passed" }, 401);
    }

    const supabaseClient = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    const { data: { user }, error: userError } = await supabaseClient.auth.getUser(
      authHeader.replace("Bearer ", "")
    );

    if (userError || !user) {
      return jsonResponse({ error: "Invalid User Token" }, 401);
    }

    // 2. Parse Request Body
    const body = await req.json();
    const {
      practiceId,
      name,
      dateOfBirth,
      gender,
      currentCountry,
      birthCountry,
      tanningBedExposure,
      previousSkinSurgeries,
      previousSunburns,
      previousCancerSurgeries,
      sunscreenUse,
      personalHistoryMelanoma,
      familyHistoryMelanoma
    } = body;

    // VALIDIERUNG: Alle Felder sind Pflichtfelder
    const requiredFields = [
      'practiceId', 'name', 'dateOfBirth', 'gender', 'currentCountry',
      'birthCountry', 'tanningBedExposure', 'previousSkinSurgeries',
      'previousSunburns', 'previousCancerSurgeries', 'sunscreenUse',
      'personalHistoryMelanoma', 'familyHistoryMelanoma'
    ];

    for (const field of requiredFields) {
      if (body[field] === undefined || body[field] === null || body[field] === "") {
        return jsonResponse({ error: `Field '${field}' is required and was not provided.` }, 400);
      }
    }

    // 3. Security Check: Validate Doctor's Practice ID
    const { data: doctor, error: doctorError } = await supabase
      .from('doctor')
      .select('practiceid')
      .eq('doctorid', user.id)
      .maybeSingle();

    if (doctorError) {
      console.error("DB Fetch Error (Doctor Validation):", doctorError);
      return jsonResponse({ error: "Database error during validation." }, 500);
    }

    if (!doctor) {
      return jsonResponse({ error: "Sender is not registered as a doctor." }, 404);
    }

    if (doctor.practiceid !== practiceId) {
      return jsonResponse({ error: "Unauthorized: Practice ID does not match sender's practice." }, 403);
    }

    // 4. Insert Data
    const { data: newPatient, error: insertError } = await supabase
      .from("patient")
      .insert({
        practiceid: practiceId,
        name: name,
        dateofbirth: dateOfBirth,
        gender: gender,
        tanningbedexposure: tanningBedExposure,
        previousskinsurgeries: previousSkinSurgeries,
        previoussunburns: previousSunburns,
        currentcountry: currentCountry,
        birthcountry: birthCountry,
        previouscancersurgeries: previousCancerSurgeries,
        sunscreenuse: sunscreenUse,
        personalhistorymelanoma: personalHistoryMelanoma,
        familyhistorymelanoma: familyHistoryMelanoma
      })
      .select()
      .single();

    if (insertError) {
      console.error("DB Insert Error:", insertError);
      return jsonResponse({
        error: "Database creation failed.",
        details: insertError.message
      }, 500);
    }

    // 5. Map DB fields to Dart Model
    const mappedPatient = {
      id: newPatient.patientid,
      name: newPatient.name,
      dateOfBirth: newPatient.dateofbirth,
      gender: newPatient.gender,
      currentCountry: newPatient.currentcountry,
      birthCountry: newPatient.birthcountry,
      tanningBedExposure: newPatient.tanningbedexposure,
      previousSkinSurgeries: newPatient.previousskinsurgeries,
      previousSunburns: newPatient.previoussunburns,
      previousCancerSurgeries: newPatient.previouscancersurgeries,
      sunscreenUse: newPatient.sunscreenuse,
      personalHistoryMelanoma: newPatient.personalhistorymelanoma,
      familyHistoryMelanoma: newPatient.familyhistorymelanoma,
      note: newPatient.note,
      practiceId: newPatient.practiceid,
    };

    return jsonResponse(mappedPatient, 201);

  } catch (e) {
    console.error("Function Error:", e);
    return jsonResponse({ error: "Internal Server Error" }, 500);
  }
});
