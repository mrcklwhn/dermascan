import { createClient } from "npm:@supabase/supabase-js@2.39.0";

// --- Environment Variables ---
const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";

if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY) {
  throw new Error("Missing Supabase configuration.");
}

// --- Supabase Client (Service Role) ---
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
  auth: {
    persistSession: false,
    autoRefreshToken: false,
    detectSessionInUrl: false,
  },
});

// --- CORS Headers ---
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// --- Helper: Response ---
const jsonResponse = (body: any, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });

Deno.serve(async (req) => {
  // 1. Handle CORS Preflight
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const authHeader = req.headers.get('Authorization');
    if (!authHeader) {
      return jsonResponse({ success: false, error: "No authorization header passed" }, 401);
    }

    const supabaseClient = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    const { data: { user }, error: userError } = await supabaseClient.auth.getUser(
      authHeader.replace("Bearer ", "")
    );

    if (userError || !user) {
      return jsonResponse({ success: false, error: "Invalid User Token" }, 401);
    }

    // 2. Parse Request Body
    const { id: lesionId } = await req.json();

    if (!lesionId) {
      return jsonResponse({ success: false, error: "Lesion ID is required." }, 400);
    }

    // 3. Fetch Data & Validation

    const { data: doctor, error: doctorError } = await supabase
      .from('doctor')
      .select('practiceid')
      .eq('doctorid', user.id)
      .maybeSingle();

    if (doctorError) {
      console.error("DB Fetch Error (Doctor):", doctorError);
      return jsonResponse({ success: false, error: "Database error fetching doctor." }, 500);
    }
    if (!doctor || !doctor.practiceid) {
      return jsonResponse({ success: false, error: "Unauthorized: Only assigned doctors can delete lesions." }, 403);
    }

    const { data: lesionInfo, error: fetchError } = await supabase
      .from('lesion')
      .select(`
        lesionid,
        patient!inner (
          practiceid
        ),
        diagnosis (
          diagnosisid
        )
      `)
      .eq('lesionid', lesionId)
      .maybeSingle();

    if (fetchError) {
      console.error("DB Fetch Error (Lesion Join):", fetchError);
      return jsonResponse({ success: false, error: "Database error fetching lesion details." }, 500);
    }
    if (!lesionInfo) {
      return jsonResponse({ success: false, error: "Lesion not found." }, 404);
    }

    const patientPracticeId = (lesionInfo as any).patient?.practiceid;
    const diagnosesCount = (lesionInfo as any).diagnosis?.length || 0;

    if (doctor.practiceid !== patientPracticeId) {
      return jsonResponse({ success: false, error: "Unauthorized: Practice mismatch." }, 403);
    }

    if (diagnosesCount > 0) {
      return jsonResponse({ success: false, error: `Deletion failed: ${diagnosesCount} diagnosis/diagnoses still exist.` }, 403);
    }

    // 4. Execute Database Operation (Delete)
    const { error: deleteError } = await supabase
      .from('lesion')
      .delete()
      .eq('lesionid', lesionId);

    if (deleteError) {
      console.error("DB Delete Error:", deleteError);
      return jsonResponse({ success: false, error: "Failed to delete lesion." }, 500);
    }

    // 5. Success Response
    return jsonResponse({
      success: true,
      message: "Lesion successfully deleted.",
      deletedId: lesionId
    }, 200);

  } catch (e) {
    console.error("Function Error:", e);
    return jsonResponse({ success: false, error: "Internal Server Error" }, 500);
  }
});
