import { createClient } from "npm:@supabase/supabase-js@2.39.0";

// --- Environment Variables ---
const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";

if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY) {
  throw new Error("Missing Supabase configuration.");
}

// --- Supabase Client (Service Role) ---
const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
  auth: {
    persistSession: false,
    autoRefreshToken: false,
    detectSessionInUrl: false,
  },
});

// --- CORS Headers ---
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// --- Helper: Response ---
const jsonResponse = (body: any, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });

Deno.serve(async (req) => {
  // 1. Handle CORS Preflight
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const authHeader = req.headers.get('Authorization');
    if (!authHeader) {
      return jsonResponse({ error: "No authorization header passed" }, 401);
    }

    const supabaseClient = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    const { data: { user }, error: userError } = await supabaseClient.auth.getUser(
      authHeader.replace("Bearer ", "")
    );

    if (userError || !user) {
      return jsonResponse({ error: "Invalid User Token" }, 401);
    }

    // 2. Parse Request Body
    const { id, date, imagePath, doctorId, lesionId, aScore, bScore, cScore, cutoutType, prediction, confidence, notes } = await req.json();

    if (!id || !date || !imagePath || !doctorId || !lesionId || aScore === undefined || bScore === undefined || cScore === undefined || !cutoutType || prediction === undefined|| !confidence) {
      return jsonResponse({ error: "All fields are required (id, date, imagePath, doctorId, lesionId, aScore, bScore, cScore, cutoutType, prediction, confidence)" }, 400);
    }

    // Security check: User must be the doctor himself
    if (user.id !== doctorId) {
      return jsonResponse({ error: "Unauthorized: You can only create diagnoses for yourself." }, 403);
    }

    // 3. Fetch Data & Validation
    // Check Doctor and his practice assignment
    const { data: doctor, error: doctorError } = await supabase
      .from('doctor')
      .select('practiceid')
      .eq('doctorid', doctorId)
      .maybeSingle();

    if (doctorError) {
      console.error("DB Fetch Error (Doctor):", doctorError);
      return jsonResponse({ error: "Database error fetching doctor." }, 500);
    }
    if (!doctor) {
      return jsonResponse({ error: "Doctor not found." }, 404);
    }
    if (!doctor.practiceid) {
      return jsonResponse({ error: "Doctor is not assigned to any practice." }, 403);
    }

    // Check Lesion and patient's practice assignment via join
    const { data: lesion, error: lesionError } = await supabase
      .from('lesion')
      .select(`
        lesionid,
        patient!inner (
          practiceid
        )
      `)
      .eq('lesionid', lesionId)
      .maybeSingle();

    if (lesionError) {
      console.error("DB Fetch Error (Lesion):", lesionError);
      return jsonResponse({ error: "Database error fetching lesion." }, 500);
    }
    if (!lesion) {
      return jsonResponse({ error: "Lesion not found." }, 404);
    }

    const patientPracticeId = (lesion.patient as any)?.practiceid;

    // Authorization Check: Ensure Doctor and Patient belong to the same practice
    if (doctor.practiceid !== patientPracticeId) {
      return jsonResponse({ error: "Unauthorized: Doctor and Patient belong to different practices." }, 403);
    }

    // 4. Execute Database Operation
    const { data: diagnosis, error: insertError } = await supabase
      .from('diagnosis')
      .insert([
        {
          diagnosisid: id,
          date: date,
          imagepath: imagePath || null,
          lesionid: lesionId,
          doctorid: doctorId,
          ascore: aScore,
          bscore: bScore,
          cscore: cScore,
          cutouttype: cutoutType,
          confidence: confidence,
          prediction: prediction,
          notes: notes
        }
      ])
      .select()
      .maybeSingle();

    if (insertError) {
      if (insertError.code === '23505') {
        return jsonResponse({ error: "Diagnosis with this ID already exists." }, 409);
      }
      console.error("DB Insert Error:", insertError);
      return jsonResponse({ error: "Failed to create diagnosis." }, 500);
    }

    if (!diagnosis) {
      return jsonResponse({ error: "Failed to retrieve created diagnosis." }, 500);
    }

    // 5. Map DB fields to Dart Model
    const mappedDiagnosis = {
      id: diagnosis.diagnosisid,
      date: diagnosis.date,
      imagePath: diagnosis.imagepath,
      lesionId: diagnosis.lesionid,
      doctorId: diagnosis.doctorid,
      aScore: diagnosis.ascore,
      bScore: diagnosis.sbcore,
      cScore: diagnosis.cscore,
      cutoutType: diagnosis.cutouttype,
      confidence: diagnosis.confidence,
      prediction: diagnosis.prediction,
      notes: diagnosis.notes
    };

    return jsonResponse(mappedDiagnosis, 201);

  } catch (e) {
    console.error("Function Error:", e);
    return jsonResponse({ error: "Internal Server Error" }, 500);
  }
});
